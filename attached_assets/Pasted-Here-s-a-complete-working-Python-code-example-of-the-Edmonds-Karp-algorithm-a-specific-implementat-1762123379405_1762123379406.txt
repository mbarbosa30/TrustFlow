Here’s a complete, working Python code example of the Edmonds-Karp algorithm (a specific implementation of Ford-Fulkerson using BFS) for maximum flow, adapted to Advogato’s trust metric model.
This simulates how Advogato used max-flow min-cut to determine which developers are trusted (“accepted”) based on a web of certifications.

Features of This Example:
	•	Uses Edmonds-Karp (BFS-based Ford-Fulkerson) — O(VE²)
	•	Models nodes with capacities (Advogato used node capacities, not just edges)
	•	Includes source → seeds, users → supersink
	•	Outputs trusted developers (those receiving ≥1 unit of flow)

Python Code: Advogato-Style Max Flow Trust Metric
from collections import defaultdict, deque
import copy

class AdvogatoMaxFlow:
    def __init__(self, num_users, seed_ids, certifications):
        """
        num_users: total number of developers (0 to num_users-1)
        seed_ids: list of trusted seed developer IDs
        certifications: list of (from_id, to_id, capacity) — trust edges
        """
        self.n = num_users
        self.seeds = seed_ids
        self.certifications = certifications

        # Node indices:
        # 0 to num_users-1: developers
        # num_users: source
        # num_users+1: supersink
        self.source = num_users
        self.sink = num_users + 1
        self.total_nodes = num_users + 2

        self.graph = defaultdict(lambda: defaultdict(int))
        self._build_graph()

    def _build_graph(self):
        # 1. Source → seeds with infinite capacity
        INF = 10**9
        for seed in self.seeds:
            self.graph[self.source][seed] = INF

        # 2. Developer → Developer: trust certifications
        for u, v, cap in self.certifications:
            self.graph[u][v] = cap  # directed edge u trusts v with capacity

        # 3. Every developer → supersink with capacity 1 (unit trust acceptance)
        for u in range(self.n):
            self.graph[u][self.sink] = 1  # each user can accept 1 unit of trust

    def bfs(self, parent):
        """BFS to find augmenting path from source to sink"""
        visited = [False] * self.total_nodes
        queue = deque()
        queue.append(self.source)
        visited[self.source] = True
        parent[self.source] = -1

        while queue:
            u = queue.popleft()
            for v in self.graph[u]:
                if not visited[v] and self.graph[u][v] > 0:
                    queue.append(v)
                    parent[v] = u
                    visited[v] = True
                    if v == self.sink:
                        return True
        return False

    def edmonds_karp(self):
        """Main max-flow algorithm"""
        parent = [-1] * self.total_nodes
        max_flow = 0
        flow = defaultdict(lambda: defaultdict(int))  # residual flow

        # Initialize residual graph as copy of original
        residual = defaultdict(lambda: defaultdict(int))
        for u in self.graph:
            for v in self.graph[u]:
                residual[u][v] = self.graph[u][v]

        while self.bfs(parent):
            # Find minimum capacity in path
            path_flow = float('inf')
            s = self.sink
            while s != self.source:
                path_flow = min(path_flow, residual[parent[s]][s])
                s = parent[s]

            # Update residual capacities
            v = self.sink
            while v != self.source:
                u = parent[v]
                residual[u][v] -= path_flow
                residual[v][u] += path_flow  # reverse edge
                v = u

            max_flow += path_flow

        # Determine which users are trusted: those who sent 1 unit to sink
        trusted_users = []
        for u in range(self.n):
            flow_to_sink = self.graph[u][self.sink] - residual[u][self.sink]
            if flow_to_sink >= 1:
                trusted_users.append(u)

        return max_flow, trusted_users, residual

# ================================
# Example Usage: Simulate Advogato
# ================================

# 6 developers: 0,1,2,3,4,5
# Seeds: 0 and 1 (trusted masters)
# Certifications: (from, to, capacity)
certifications = [
    (0, 2, 5), (0, 3, 3),
    (1, 2, 4), (1, 4, 6),
    (2, 3, 2), (2, 5, 3),
    (3, 5, 1),
    (4, 5, 2)
]

num_users = 6
seeds = [0, 1]

# Create Advogato trust network
advogato = AdvogatoMaxFlow(num_users, seeds, certifications)

# Run max-flow
total_trust_flow, trusted_developers, residual = advogato.edmonds_karp()

print(f"Total trust flow (max flow): {total_trust_flow}")
print(f"Trusted developers (accepted into community): {trusted_developers}")
print("(These are the ones who received at least 1 unit of trust flow)")

# Optional: Show flow from each user to sink
print("\nFlow from each developer to supersink:")
for u in range(num_users):
    flow_used = 1 - residual[u][num_users + 1]  # original was 1
    print(f"  User {u}: {flow_used}/1 unit used → {'TRUSTED' if flow_used >= 1 else 'not trusted'}")
