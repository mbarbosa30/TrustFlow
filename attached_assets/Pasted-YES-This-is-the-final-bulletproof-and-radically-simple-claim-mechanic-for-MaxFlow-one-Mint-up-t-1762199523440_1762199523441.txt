Final Claim Mechanics: `claim() → mint(score)` with Hard Cap + Cooldown
uint256 constant DAILY_MINT_CAP = 1000;
uint256 constant CLAIM_COOLDOWN = 7 days;
uint256 public lastMintDay;
uint256 public mintedToday;

mapping(address => uint256) public lastClaim;

function claim() external {
    require(block.timestamp >= lastClaim[msg.sender] + CLAIM_COOLDOWN, "7-day cooldown");
    
    uint256 score = oracle.getLocalHealth(msg.sender);
    require(score >= 10, "Score too low");

    // Reset daily counter at midnight UTC
    uint256 today = block.timestamp / 1 days;
    if (today > lastMintDay) {
        mintedToday = 0;
        lastMintDay = today;
    }

    uint256 maxMint = score;  // or (score * score) / 100 for power law
    uint256 available = DAILY_MINT_CAP - mintedToday;
    
    uint256 mintAmount = min(maxMint, available);
    require(mintAmount > 0, "Daily cap reached");

    mintedToday += mintAmount;
    lastClaim[msg.sender] = block.timestamp;
    
    trustToken.mint(msg.sender, mintAmount);
}
