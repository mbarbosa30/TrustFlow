5. Scoring Architecture (Final)
// 1. V-Score (unchanged)
V_Score = percentile(residual_flow)

// 2. T-Score (new)
net_received = sum(received) - sum(sent)  // 30-day
unique_senders = count(distinct senders)
sender_avg_vscore = avg(V_Score of senders)

raw_t = 0.5 * clamp(net_received / max_net, 0, 1)
        + 0.3 * (unique_senders / max_senders)
        + 0.2 * sender_avg_vscore

T_Score = percentile(raw_t)

// 3. Final
Final_Score = 0.7 * V_Score + 0.3 * T_Score
