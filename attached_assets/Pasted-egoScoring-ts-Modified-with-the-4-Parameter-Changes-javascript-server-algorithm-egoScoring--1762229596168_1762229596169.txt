function calculateLocalHealth(metrics) {
  const {
    avgResidualFlow,
    medianMinCut,
    seedCount,
    egoSize,
    edgeCount,
    outgoingVouches,
    kudosWeight,  // Assume this is the total decayed KUDOS on edges
    // ... other params
  } = metrics;

  // Change 1: Exponential scaling on components
  const saturation = avgResidualFlow;  // From direct flow / max inbound
  const flowComponent = 70 * (saturation ** 1.2);

  const redundancy = medianMinCut / seedCount;  // Or your current redundancy calculation
  const redundancyComponent = 40 * (redundancy ** 1.2);

  // Change 2: Raise KUDOS threshold to 500
  const boostMultiplier = 1 + Math.min(1, kudosWeight / 500);

  // Change 3: Strengthen dilution to 0.1 per excess vouch
  let dilutionFactor = 1.0;
  if (outgoingVouches > 10) {
    const excess = outgoingVouches - 10;
    dilutionFactor = Math.max(0.5, 1 - 0.1 * excess);  // 0.1 per excess
  }

  // Apply dilution to redundancy
  const adjustedRedundancyComponent = redundancyComponent * dilutionFactor;

  // Total score
  const localHealth = flowComponent + adjustedRedundancyComponent;
  return Math.min(100, Math.max(0, localHealth));
}

### Implementation Notes
- **Exponential Scaling**: **1.2** gives diminishing high-end (1.0**1.2=1.0, 0.5**1.2=0.43—pulls mid-low down for spread). Tune to 1.5 for steeper (0.5**1.5=0.35).
- **KUDOS Threshold**: /500 means 250 KUDOS for 0.5 boost, 500 for max 1.0—nudge, not lever (+6 pts max for 500).
- **Dilution**: 0.1 per excess = stronger bite (15=0.5 factor = -20 pts redundancy = -8 total from 97.5 to 89.5).
- **Robustness**: Heuristics reduced (baseline from graph, no /20); quality mean from residuals (computed in flow run).
- **Fairness**: Merit-focused (depth/quality drives redundancy); dilution prevents spam.
