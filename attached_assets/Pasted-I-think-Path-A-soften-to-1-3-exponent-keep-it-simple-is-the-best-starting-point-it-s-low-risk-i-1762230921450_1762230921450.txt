function calculateLocalHealth(metrics) {
  const {
    avgResidualFlow,  // Saturation from direct_flow / max_inbound
    medianMinCut,  // For redundancy
    seedCount,  // Or vouch_count
    egoSize,
    edgeCount,
    outgoingVouches,
    kudosWeight,
  } = metrics;

  // Path A: Soften exponential to 1.3
  const saturation = avgResidualFlow;
  const flowComponent = 70 * Math.pow(saturation, 1.3);

  const redundancy = medianMinCut / seedCount;  // Or effective / baseline
  const redundancyComponent = 40 * Math.pow(redundancy, 1.3);

  // Raise KUDOS threshold to 500
  const boostMultiplier = 1 + Math.min(1, kudosWeight / 500);

  // Strengthen dilution to 0.1 per excess
  let dilutionFactor = 1.0;
  if (outgoingVouches > 10) {
    const excess = outgoingVouches - 10;
    dilutionFactor = Math.max(0.5, 1 - 0.1 * excess);
  }

  // Adjusted
  const adjustedRedundancyComponent = redundancyComponent * dilutionFactor;

  const localHealth = Math.min(100, Math.max(0, flowComponent + adjustedRedundancyComponent));
  return localHealth;
}
